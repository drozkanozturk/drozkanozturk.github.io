<script>
(function(){
  var D=["","MON","TUE","WED","THU","FRI"];
  function m(t){var p=t.split(":");return parseInt(p[0],10)*60+parseInt(p[1],10);}
  function f(t){var p=t.split(":");return parseInt(p[0],10)+":"+p[1];}
  var S=8*60, E=20*60, STEP=30, rows=(E-S)/STEP;
  var b='border:1px solid rgba(201,168,76,0.18);';
  var skip=[0,0,0,0,0,0];
  var h='<thead><tr><th style="'+b+'width:2.4rem;"></th>';
  for(var d=1;d<=5;d++) h+='<th style="'+b+'padding:0.35rem 0.2rem;color:var(--gold);font-weight:400;letter-spacing:0.08em;">'+D[d]+'</th>';
  h+='</tr></thead><tbody>';
  for(var r=0;r<rows;r++){
    var t=S+r*STEP;
    h+='<tr>';
    if(r%2===0) h+='<td rowspan="2" style="'+b+'padding:0.15rem 0.3rem;color:var(--steel);vertical-align:top;text-align:right;">'+(t/60)+':00</td>';
    for(var d=1;d<=5;d++){
      if(skip[d]>0){skip[d]--;continue;}
      var ev=null;
      EVENTS.forEach(function(e){ if(e.day===d && m(e.start)===t) ev=e; });
      if(ev){
        var span=Math.max(1,Math.round((m(ev.end)-m(ev.start))/STEP));
        skip[d]=span-1;
        var meet=ev.type==="meet";
        var col=meet?"#7eb8d4":"var(--gold-light)";
        var bg=meet?"rgba(100,160,200,0.15)":"rgba(201,168,76,0.18)";
        h+='<td rowspan="'+span+'" style="'+b+'padding:0.25rem 0.3rem;vertical-align:top;color:'+col+';background:'+bg+';">'+ev.title+'<br><span style="opacity:.75;">'+f(ev.start)+'–'+f(ev.end)+'</span></td>';
      } else {
        h+='<td style="'+b+'height:1.1rem;"></td>';
      }
    }
    h+='</tr>';
  }
  h+='</tbody>';
  document.getElementById("mini-schedule").innerHTML=h;
})();
</script>
